name: Sync Vietlott Live Data

on:
  schedule:
    - cron: '0 12 * * *'
  workflow_dispatch:

permissions:
  contents: write

jobs:
  sync:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout repo
        uses: actions/checkout@v4

      - name: Cài đặt Python
        uses: actions/setup-python@v5
        with:
          python-version: '3.x'

      - name: Cài đặt thư viện
        run: pip install requests

      - name: Tải dữ liệu Vietlott
        run: python scripts/fetch_vietlott.py

      - name: Cập nhật dữ liệu vào Repo
        run: |
          git config --global user.name "github-actions[bot]"
          git config --global user.email "github-actions[bot]@users.noreply.github.com"
          git add -A
          if ! git diff --staged --quiet; then
            git commit -m "Auto-update Vietlott live data"
            git push
          else
            echo "Dữ liệu không có thay đổi."
          fi
